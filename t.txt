onClick={handleExportVisitorsPDF}>Export Visitors PDF</Button>            </div>            {(() => {              const rows = getDailyStatsInRange(dateRange.start || undefined, dateRange.end || undefined)              if (rows.length === 0) {                return <div className="text-center py-8 text-muted-foreground">Belum ada data visitor</div>              }              // Show per-day cards;
 click to open detail dialog              return rows.map(r => (                <Card key={r.date} className="cursor-pointer hover:bg-accent/50 transition-colors"                  onClick={() => {                    const lostLogs = getFromLS<any[]>(LS_KEYS.VISITOR_LOST_LOG, []);
                    const notes = lostLogs.filter(e => e.date === r.date).map(e => e.description);
                    setVisitorDetail({ date: r.date, notes });
                  }}>                  <CardContent className="p-4 flex items-center justify-between">                    <div>                      <div className="font-medium">{new Date(r.date).toLocaleDateString('id-ID')}</div>                      <div className="text-xs text-muted-foreground">Ringkasan tamu harian</div>                    </div>                    <div <span className="font-semibold">{r.lost}</span></div>                    </div>                  </CardContent>                </Card>              ))            })()}          </TabsContent>        </Tabs>      </div>      {/* Transaction Detail Dialog */}      <Dialog open={isDialogOpen} onOpenChange={setIsDialogOpen}>        <DialogContent className="w-full max-w-[98vw] sm:max-w-md p-2 sm:p-6">          <DialogHeader>            <DialogTitle>Detail Transaksi</DialogTitle>          </DialogHeader>          {selectedTransaction && (            <div className="space-y-4">              <div className="flex justify-between items-center">                <div>                  <p className="font-medium">{selectedTransaction.id}</p>                  <p className="text-sm            </p>                </div>                <Badge                  variant={getStatusBadgeVariant(selectedTransaction.status)}                  className="mt-1"                > "Batal"}                  {selectedTransaction.status === "refunded" && "Refund"}                </Badge>              </div>              <div>                <p className="font-medium">Pelanggan</p>                <p>{selectedTransaction.customer}</p>              </div>              <p className="font-medium mb-2">Item</p>              <div className="space-y-2">                {selectedTransaction.items.map((item, index) => {                  // Ambil produk dari localStorage                  const allProducts = getFromLS(LS_KEYS.PRODUCTS, DUMMY_PRODUCTS);
                  let sku = item.sku;
                  if (!sku) {                    const prod = allProducts.find(p => p.name === item.name);
                    sku = prod?.sku || "-";

                })}              </div>              <Separator />              <div className="flex justify-between items-center">                <p className="font-bold">Total</p>                <p className="font-bold text-amber-600">                  {formatCurrency(selectedTransaction.total)}                </p>              </div>              <div className="flex gap-2 pt-4">                <Button                  variant="outline"                  className="flex-1 flex items-center justify-center gap-2"                  onClick={printReceipt}                >                  <Printer className="h-4 w-4" />                  Cetak Struk                </Button>                {selectedTransaction.status === "completed" && (                  <Button                    variant="outline"                    className="flex-1 flex items-center justify-center gap-2"                    onClick={createReturnTransaction}                  >                    <RotateCcw className="h-4 w-4" />                    Retur                  </Button>                )}              </div>            </div>          )}        </DialogContent>      </Dialog>    </div>  );
